getInstanceName() の束縛と、成分配列の if 方程式で EF004

原因は 2 つで、コードが同じ。v0.10.4 でも両方とも再現する

// A: parameter の束縛
model Rep21A
  final parameter String name =
    getInstanceName();          // EF004
end Rep21A;

// B: 成分配列 × ベクトル（スカラ 1 行）
model Rep21B
  parameter Integer n = 2;
  Rep21Rec s[n];
equation
  if noEvent(q > 0.5) then
    hMix*(s.n_flow*ones(n)) = 0; // EF004
  else
    hMix = 0;
  end if;
end Rep21B;
