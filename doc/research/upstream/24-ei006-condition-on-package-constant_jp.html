条件付き成分の条件が別の package の定数を読むと EI006

package の定数からできた条件を parameter 式と認めない。v0.10.4 でも同じ

package Medium
  constant Integer nX = 2;
  constant Integer nXi = nX - 1;
end Medium;

model C
  Real x = time if Medium.nX > 1; // EI006
end C;

model C0  // 対照: モデルの定数なら通る
  constant Integer nX = 2;
  Real x = time if nX > 1;
end C0;
